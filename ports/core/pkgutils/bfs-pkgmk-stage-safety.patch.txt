+# Scoped fakeroot for install targets in classic one-function recipes.
+# This does NOT wrap configure/compile; this may need further handlers for
+# non-make staging commands, which must fail closed instead of publishing.
+_bfs_universal_make() {
+    local arg stage=no
+    for arg in "$@"; do
+        case "$arg" in install|install-*|*-install|install/strip) stage=yes ;; esac
+    done
+    if [ "$stage" = yes ]; then
+        info "BFSOS: scoped fakeroot make install: $name"
+        timeout --signal=TERM --kill-after=10s "${BFS_PKG_STAGE_TIMEOUT:-1800}" \
+            fakeroot -- make "$@"
+    else
+        command make "$@"
+    fi
+}
